        // Validamos si debemos ocultar el botón masivo
        if (data && data.mostrar_boton_varias === false) {
            $("#gestionar-marca-masivo").hide();
        }

            // Obtenemos las marcas actualizadas del servidor
            const marcasActualizadas = await ObtenerMarcasAPI();

            // Actualizamos la variable global de window
            window.marcas = marcasActualizadas;

            // Re-pintamos el custom select con las nuevas marcas
            PintarSelectMarcas();

        } catch (error) {
            console.error("Error al actualizar las marcas en el padre:", error);
        }

        // Cerramos el modal de forma controlada
        if (typeof ModalAPI !== 'undefined' && typeof ModalAPI.cerrar === 'function') {
            setTimeout(() => {
                ModalAPI.cerrar();
            }, 1000);
        }
    });
});

// Función para actualizar el costo base global
function actualizarCostoBaseGlobal() {
    let costoTotal = parseFloat($('#producto-input-costo-total').val()) || 0;
    let piezas = parseInt($('#producto-input-piezas-compra').val()) || 1;

    costoTotalCompra = costoTotal;
    piezasTotalesCompra = piezas > 0 ? piezas : 1;
    costoBaseUnitarioGlobal = costoTotalCompra / piezasTotalesCompra;

    $('#lbl-costo-base-unitario').text(`$${costoBaseUnitarioGlobal.toFixed(4)}`);

    // Recalcular filas que estén vinculadas
    presentacionesVenta.forEach((item, index) => {
        if (item.vinculado) {
            item.costoUnitario = costoBaseUnitarioGlobal * item.equivalencia;
            let nuevoPrecio = item.costoUnitario * (1 + (item.gananciaPorcentaje / 100));
            item.precioVenta = parseFloat(nuevoPrecio.toFixed(2));
        }
    });
    renderizarTablaPreciosVenta();
}

// Evento para alternar entre Vinculado y Desvinculado con el botón de cadena
$(document).on('click', '.btn-toggle-vinculo', function () {
    const index = $(this).data('index');
    let item = presentacionesVenta[index];

    // Alternar estado
    item.vinculado = !item.vinculado;

    if (item.vinculado) {
        // Si pasa a estar vinculado, se recalcula automáticamente con base en el costo global y equivalencia
        item.costoUnitario = costoBaseUnitarioGlobal * item.equivalencia;
        let nuevoPrecio = item.costoUnitario * (1 + (item.gananciaPorcentaje / 100));
        item.precioVenta = parseFloat(nuevoPrecio.toFixed(2));
    }

    renderizarTablaPreciosVenta();
});

// Eventos interactivos en la tabla de Precios y Costos (Paso 5)
$(document).on('input', '.input-empaque, .input-equiv, .input-costo-unitario, .input-ganancia, .input-precio-venta', function () {
    const index = $(this).data('index');
    const valor = $(this).val();
    let item = presentacionesVenta[index];

    if ($(this).hasClass('input-empaque')) {
        item.empaque = valor;
        renderizarTablaCodigosSat();
    }
    if ($(this).hasClass('input-equiv')) {
        item.equivalencia = parseInt(valor) || 0;
        if (item.vinculado) {
            item.costoUnitario = costoBaseUnitarioGlobal * item.equivalencia;
            let nuevoPrecio = item.costoUnitario * (1 + (item.gananciaPorcentaje / 100));
            item.precioVenta = parseFloat(nuevoPrecio.toFixed(2));
        }
        renderizarTablaCodigosSat();
        renderizarTablaPreciosVenta();
    }
    if ($(this).hasClass('input-costo-unitario') && !item.vinculado) {
        // Edición manual de costo (solo permitido si está desvinculado)
        item.costoUnitario = parseFloat(valor) || 0.00;
        let nuevoPrecio = item.costoUnitario * (1 + (item.gananciaPorcentaje / 100));
        item.precioVenta = parseFloat(nuevoPrecio.toFixed(2));
        $(`.input-precio-venta[data-index="${index}"]`).val(item.precioVenta.toFixed(2));
    }
    if ($(this).hasClass('input-ganancia')) {
        item.gananciaPorcentaje = parseFloat(valor) || 0;
        let nuevoPrecio = item.costoUnitario * (1 + (item.gananciaPorcentaje / 100));
        item.precioVenta = parseFloat(nuevoPrecio.toFixed(2));
        $(`.input-precio-venta[data-index="${index}"]`).val(item.precioVenta.toFixed(2));
    }
    if ($(this).hasClass('input-precio-venta')) {
        item.precioVenta = parseFloat(valor) || 0.00;
        if (item.costoUnitario > 0) {
            let margenCalculado = ((item.precioVenta - item.costoUnitario) / item.costoUnitario) * 100;
            item.gananciaPorcentaje = parseFloat(margenCalculado.toFixed(2));
            $(`.input-ganancia[data-index="${index}"]`).val(item.gananciaPorcentaje);
        }
    }
});

// Eventos para la sección independiente de Códigos y SAT (Paso 6)
$(document).on('input', '.input-codigo, .input-sat', function () {
    const index = $(this).data('index');
    const valor = $(this).val();

    if ($(this).hasClass('input-codigo')) {
        presentacionesVenta[index].codigoBarras = valor;
    }
    if ($(this).hasClass('input-sat')) {
        presentacionesVenta[index].claveSat = valor;
    }
});
